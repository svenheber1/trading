//| RISICO-FIX                                                        |
//| SYMBOL_TRADE_TICK_VALUE klopt niet in backtests op custom         |
//| symbolen. Deze functies laten MT5 de waarde zelf uitrekenen met   |
//| OrderCalcProfit(), met de koers van dat moment. Live identiek.    |
//+------------------------------------------------------------------+
double RF_WaardePerPrijs(string sym)
  {
   double bid = SymbolInfoDouble(sym, SYMBOL_BID);
   double ts  = SymbolInfoDouble(sym, SYMBOL_TRADE_TICK_SIZE);
   if(bid > 0 && ts > 0)
     {
      double afstand = MathMin(1000.0 * ts, bid * 0.5);
      double verlies = 0.0;
      if(OrderCalcProfit(ORDER_TYPE_BUY, sym, 1.0, bid, bid - afstand, verlies) && verlies < 0)
         return(-verlies / afstand);
     }
   // noodgreep: oude waarde (alleen als OrderCalcProfit niet lukt)
   double tv = SymbolInfoDouble(sym, SYMBOL_TRADE_TICK_VALUE);
   return(ts > 0 ? tv / ts : 0.0);
  }

// waarde van 1 lot per TICK (vervangt SYMBOL_TRADE_TICK_VALUE)
double RF_TickWaarde(string sym)
  {
   return(RF_WaardePerPrijs(sym) * SymbolInfoDouble(sym, SYMBOL_TRADE_TICK_SIZE));
  }

// waarde van 1 lot per PUNT (Profectus rekent de stop in punten)
double RF_PuntWaarde(string sym)
  {
   return(RF_WaardePerPrijs(sym) * SymbolInfoDouble(sym, SYMBOL_POINT));
  }
//+--- einde RISICO-FIX ----------------------------------------------+
